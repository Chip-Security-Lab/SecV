module acct_wrapper #(
    parameter AcCt_MEM_SIZE = 8
) (
    input wire clk_i,
    input wire rst_ni,
    input wire rst_6,
    output wire [AcCt_MEM_SIZE * 32 - 1 : 0] acct_mem
);
    initial begin
        for (integer j = 0; j < AcCt_MEM_SIZE; j = j+1) begin
            acct_mem[j*32 +: 32] = 32'h00000000;  // Case 1: Default Permissions Disabled
            // acct_mem[j*32 +: 32] = 32'hFFFFFFFF; // Case 2: Default Permissions Enabled
        end
    end

    always@(posedge clk_i or negedge rst_ni) begin
        if(~rst_ni && ~rst_6) begin
            for (integer j = 0; j < AcCt_MEM_SIZE; j = j+1) begin
                acct_mem[j*32 +: 32] = 32'h00000000;  // Case 1: Default Permissions Disabled
                // acct_mem[j*32 +: 32] = 32'hFFFFFFFF; // Case 2: Default Permissions Enabled
            end
        end
    end
endmodule
