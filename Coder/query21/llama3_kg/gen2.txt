module acct_wrapper #(
    parameter AcCt_MEM_SIZE = 8 
)(
    input  wire                 clk_i,    
    input  wire                 rst_ni,   
    input  wire                 rst_6,    
    output reg [AcCt_MEM_SIZE*32-1:0] acct_mem  
);

    integer j;

    always @(posedge clk_i) begin
        if (~(rst_ni && ~rst_6)) begin
            for (j = 0; j < AcCt_MEM_SIZE; j = j + 1) begin
                acct_mem[j*32 +: 32] <= 32'h00000000; 
            end
        end
    end
endmodule
