module acct_wrapper(
    input wire clk_i,
    input wire rst_ni,
    input wire rst_6,
    output reg [AcCt_MEM_SIZE*32-1:0] acct_mem
);

parameter AcCt_MEM_SIZE = 8;

always @(posedge clk_i) begin
    if (!(rst_ni && ~rst_6)) begin
        // Case 1: Default Permissions Disabled
        // acct_mem = {AcCt_MEM_SIZE{32'h00000000}};

        // Case 2: Default Permissions Enabled
        acct_mem = {AcCt_MEM_SIZE{32'hFFFFFFFF}};
    end
end

endmodule
