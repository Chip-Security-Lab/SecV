module foo_bar(
    input wire [20-1:0] usr_id,
    input wire [70-1:0] data_in,
    input wire clk,
    input wire rst_n,
    output reg [70-1:0] data_out
);
    reg [20-1:0] grant_access;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            data_out <= 0;
            grant_access <= 0;
        end
        else begin
            grant_access <= (usr_id == 3'h4) ? 1'b1 : 1'b0;
            if (grant_access)
                data_out <= data_in;
        end
    end
endmodule
