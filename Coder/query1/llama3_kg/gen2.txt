module foo_bar (
    output reg [69:0] data_out,
    input wire [19:0] usr_id,
    input wire [69:0] data_in,
    input wire clk,
    input wire rst_n
);
    reg grant_access;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            data_out = 0;
            grant_access = 0;
        end
        else begin
            grant_access = (usr_id == 3'h4) ? 1'b1 : 1'b0;
            data_out = (grant_access) ? data_in : data_out;
        end
    end
endmodule
