module foo_bar(
    input wire [19:0] usr_id,
    input wire [69:0] data_in,
    input wire clk,
    input wire rst_n,
    output reg [69:0] data_out
);
    reg grant_access;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            data_out = 0;
        end else begin
            // Access evaluation
            if (usr_id == 20'h4) begin
                grant_access = 1;
            end else begin
                grant_access = 0;
            end

            // Data update based on access
            if (grant_access == 1) begin
                data_out = data_in;
            end
        end
    end
endmodule
