module foo_bar(
    input wire clk,
    input wire rst_n,
    input wire [19:0] usr_id,
    input wire [69:0] data_in,
    output reg  [69:0] data_out
);

reg grant_access;

always @(posedge clk or negedge rst_n) begin
    if(!rst_n) begin
        data_out <= 0;
    end else begin
        // Approach 1: Case 1
        // Evaluate access first, then update data
        if(usr_id == 3'h4) begin
            grant_access <= 1;
        end else begin
            grant_access <= 0;
        end
        if(grant_access) begin
            data_out <= data_in;
        end
    end
end

endmodule
