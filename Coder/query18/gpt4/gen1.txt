module reglk_wrapper #(
  parameter REG_COUNT = 6
) (
  input wire clk_i,
  input wire rst_ni,
  input wire jtag_unlock,
  input wire rst_9,
  output wire [31:0] reglk_mem0,
  output wire [31:0] reglk_mem1,
  output wire [31:0] reglk_mem2,
  output wire [31:0] reglk_mem3,
  output wire [31:0] reglk_mem4,
  output wire [31:0] reglk_mem5
);

  reg [31:0] reglk_mem [0:REG_COUNT-1];

  // Set the initial value to fully locked state (YOUR_OPTION = '1')
  // Set the initial value to fully unlocked state (YOUR_OPTION = '0')

  initial begin
    reglk_mem[0] = YOUR_OPTION ? 32'hFFFFFFFF : 32'h00000000;
    reglk_mem[1] = YOUR_OPTION ? 32'hFFFFFFFF : 32'h00000000;
    reglk_mem[2] = YOUR_OPTION ? 32'hFFFFFFFF : 32'h00000000;
    reglk_mem[3] = YOUR_OPTION ? 32'hFFFFFFFF : 32'h00000000;
    reglk_mem[4] = YOUR_OPTION ? 32'hFFFFFFFF : 32'h00000000;
    reglk_mem[5] = YOUR_OPTION ? 32'hFFFFFFFF : 32'h00000000;
  end

  always @(posedge clk_i or negedge rst_ni) begin
    if (~rst_ni || jtag_unlock || ~rst_9) begin
      reglk_mem[0] <= YOUR_OPTION ? 32'hFFFFFFFF : 32'h00000000;
      reglk_mem[1] <= YOUR_OPTION ? 32'hFFFFFFFF : 32'h00000000;
      reglk_mem[2] <= YOUR_OPTION ? 32'hFFFFFFFF : 32'h00000000;
      reglk_mem[3] <= YOUR_OPTION ? 32'hFFFFFFFF : 32'h00000000;
      reglk_mem[4] <= YOUR_OPTION ? 32'hFFFFFFFF : 32'h00000000;
      reglk_mem[5] <= YOUR_OPTION ? 32'hFFFFFFFF : 32'h00000000;
    end else begin
      // Update logic
    end
  end

  assign reglk_mem0 = reglk_mem[0];
  assign reglk_mem1 = reglk_mem[1];
  assign reglk_mem2 = reglk_mem[2];
  assign reglk_mem3 = reglk_mem[3];
  assign reglk_mem4 = reglk_mem[4];
  assign reglk_mem5 = reglk_mem[5];

endmodule
