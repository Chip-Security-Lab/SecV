module reglk_wrapper #(
    parameter REG_COUNT = 6
)(
    input wire clk_i,
    input wire rst_ni,
    input wire jtag_unlock,
    input wire rst_9,
    output wire [31:0] reglk_mem0,
    output wire [31:0] reglk_mem1,
    output wire [31:0] reglk_mem2,
    output wire [31:0] reglk_mem3,
    output wire [31:0] reglk_mem4,
    output wire [31:0] reglk_mem5
);

    // Register definitions
    reg [31:0] _reglk_mem0 = 32'hFFFFFFFF;
    reg [31:0] _reglk_mem1 = 32'hFFFFFFFF;
    reg [31:0] _reglk_mem2 = 32'hFFFFFFFF;
    reg [31:0] _reglk_mem3 = 32'hFFFFFFFF;
    reg [31:0] _reglk_mem4 = 32'hFFFFFFFF;
    reg [31:0] _reglk_mem5 = 32'hFFFFFFFF;

    assign reglk_mem0 = _reglk_mem0;
    assign reglk_mem1 = _reglk_mem1;
    assign reglk_mem2 = _reglk_mem2;
    assign reglk_mem3 = _reglk_mem3;
    assign reglk_mem4 = _reglk_mem4;
    assign reglk_mem5 = _reglk_mem5;

    // Reset logic
    always @(posedge clk_i or negedge rst_ni or posedge jtag_unlock or negedge rst_9) begin
        if (!rst_ni || jtag_unlock || !rst_9) begin
            _reglk_mem0 <= 32'hFFFFFFFF; // for locked status, use 0 for free status
            _reglk_mem1 <= 32'hFFFFFFFF;
            _reglk_mem2 <= 32'hFFFFFFFF;
            _reglk_mem3 <= 32'hFFFFFFFF;
            _reglk_mem4 <= 32'hFFFFFFFF;
            _reglk_mem5 <= 32'hFFFFFFFF;
        end
        // other logic can be added here if needed
    end
endmodule
