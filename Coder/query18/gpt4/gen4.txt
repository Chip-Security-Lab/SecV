(
  input wire clk_i,
  input wire rst_ni,
  input wire jtag_unlock,
  input wire rst_9,

  output wire [31:0] reglk_mem1,
  output wire [31:0] reglk_mem2,
  output wire [31:0] reglk_mem3,
  output wire [31:0] reglk_mem4,
  output wire [31:0] reglk_mem5
);

  // localparam INIT_VALUE = 32'hFFFFFFFF;
  // Case 2: Fully Unlocked State

  always @(posedge clk_i or negedge rst_ni or posedge jtag_unlock or negedge rst_9) begin
