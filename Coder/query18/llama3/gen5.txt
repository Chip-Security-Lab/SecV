module reglk_wrapper #(
    parameter REG_COUNT = 6
) (
    input wire clk_i,
    input wire rst_ni,
    input wire jtag_unlock,
    input wire rst_9,
    output reg [31:0] reglk_mem0,
    output reg [31:0] reglk_mem1,
    output reg [31:0] reglk_mem2,
    output reg [31:0] reglk_mem3,
    output reg [31:0] reglk_mem4,
    output reg [31:0] reglk_mem5
);

    // Reset initialization logic (Case 1: Fully Locked State)
    // Uncomment the following lines for a fully locked state by default
    // assign reglk_mem0 = (rst_ni && !jtag_unlock && !rst_9) ? reglk_mem0 : 32'hFFFFFFFF;
    // assign reglk_mem1 = (rst_ni && !jtag_unlock && !rst_9) ? reglk_mem1 : 32'hFFFFFFFF;
    // assign reglk_mem2 = (rst_ni && !jtag_unlock && !rst_9) ? reglk_mem2 : 32'hFFFFFFFF;
    // assign reglk_mem3 = (rst_ni && !jtag_unlock && !rst_9) ? reglk_mem3 : 32'hFFFFFFFF;
    // assign reglk_mem4 = (rst_ni && !jtag_unlock && !rst_9) ? reglk_mem4 : 32'hFFFFFFFF;
    // assign reglk_mem5 = (rst_ni && !jtag_unlock && !rst_9) ? reglk_mem5 : 32'hFFFFFFFF;

    // Reset initialization logic (Case 2: Fully Unlocked State)
    // Uncomment the following lines for a fully unlocked state by default
    assign reglk_mem0 = (rst_ni && !jtag_unlock && !rst_9) ? reglk_mem0 : 32'h00000000;
    assign reglk_mem1 = (rst_ni && !jtag_unlock && !rst_9) ? reglk_mem1 : 32'h00000000;
    assign reglk_mem2 = (rst_ni && !jtag_unlock && !rst_9) ? reglk_mem2 : 32'h00000000;
    assign reglk_mem3 = (rst_ni && !jtag_unlock && !rst_9) ? reglk_mem3 : 32'h00000000;
    assign reglk_mem4 = (rst_ni && !jtag_unlock && !rst_9) ? reglk_mem4 : 32'h00000000;
    assign reglk_mem5 = (rst_ni && !jtag_unlock && !rst_9) ? reglk_mem5 : 32'h00000000;

    // Synchronous update of lock registers
    always @(posedge clk_i) begin
        if (!rst_ni || jtag_unlock || rst_9) begin
            reglk_mem0 <= (rst_ni && !jtag_unlock && !rst_9) ? reglk_mem0 : 32'h00000000;
            reglk_mem1 <= (rst_ni && !jtag_unlock && !rst_9) ? reglk_mem1 : 32'h00000000;
            reglk_mem2 <= (rst_ni && !jtag_unlock && !rst_9) ? reglk_mem2 : 32'h00000000;
            reglk_mem3 <= (rst_ni && !jtag_unlock && !rst_9) ? reglk_mem3 : 32'h00000000;
            reglk_mem4 <= (rst_ni && !jtag_unlock && !rst_9) ? reglk_mem4 : 32'h00000000;
            reglk_mem5 <= (rst_ni && !jtag_unlock && !rst_9) ? reglk_mem5 : 32'h00000000;
        end
    end

endmodule
