module reglk_wrapper #(
    parameter REG_COUNT = 6
) (
    input  wire        clk_i,
    input  wire        rst_ni,
    input  wire        jtag_unlock,
    input  wire        rst_9,
    output reg [31:0]  reglk_mem0,
    output reg [31:0]  reglk_mem1,
    output reg [31:0]  reglk_mem2,
    output reg [31:0]  reglk_mem3,
    output reg [31:0]  reglk_mem4,
    output reg [31:0]  reglk_mem5
);

// Case 1: Fully Locked State
localparam INIT_LOCKED = 32'hFFFFFFFF;

// Case 2: Fully Unlocked State
// localparam INIT_UNLOCKED = 0;

always @(posedge clk_i) begin
    if (~rst_ni || jtag_unlock || ~rst_9) begin
        reglk_mem0 <= INIT_LOCKED;
        reglk_mem1 <= INIT_LOCKED;
        reglk_mem2 <= INIT_LOCKED;
        reglk_mem3 <= INIT_LOCKED;
        reglk_mem4 <= INIT_LOCKED;
        reglk_mem5 <= INIT_LOCKED;
    end
end

endmodule
