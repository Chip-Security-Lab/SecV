module reglk_wrapper #(
    parameter REG_COUNT = 6
) (
    input  wire clk_i,
    input  wire rst_ni,
    input  wire jtag_unlock,
    input  wire rst_9,
    output reg [31:0] reglk_mem0,
    output reg [31:0] reglk_mem1,
    output reg [31:0] reglk_mem2,
    output reg [31:0] reglk_mem3,
    output reg [31:0] reglk_mem4,
    output reg [31:0] reglk_mem5
);

    // Case 1: Fully Locked State (Comment out for Case 2)
    localparam LOCKED_STATE = 32'hFFFFFFFF;
    // localparam LOCKED_STATE = 32'h00000000; // Uncomment for Case 2: Fully Unlocked State

    always @(posedge clk_i) begin
        if (~rst_ni || jtag_unlock || ~rst_9) begin
            reglk_mem0 <= LOCKED_STATE;
            reglk_mem1 <= LOCKED_STATE;
            reglk_mem2 <= LOCKED_STATE;
            reglk_mem3 <= LOCKED_STATE;
            reglk_mem4 <= LOCKED_STATE;
            reglk_mem5 <= LOCKED_STATE;
        end
    end

endmodule
