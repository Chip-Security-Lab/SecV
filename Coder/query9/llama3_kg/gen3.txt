module sft(
    input wire clk,
    input wire rst,
    input wire [3:0] a,
    output reg [3:0] q
);

parameter MODE = 0; // 0: Serial, 1: Parallel

reg [3:0] temp_q;

always @(posedge clk, posedge rst)
begin
    if (rst == 1'b1)
        temp_q <= 4'b0000;
    else if (MODE == 0) // Serial Data Transfer
        temp_q <= {temp_q[2:0], a[3]};
    else // Parallel Data Transfer
        temp_q <= a;
end

assign q = temp_q;

endmodule
