module sft(
    input wire clk,
    input wire rst,
    input wire [3:0] a,
    output reg [3:0] q
);

    // Serial Data Transfer
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            q <= 4'b0000;
        end else begin
            q <= q << 1;
            q[0] <= a[0];
        end
    end

    // Parallel Data Transfer
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            q <= 4'b0000;
        end else begin
            q <= a;
        end
    end
endmodule
