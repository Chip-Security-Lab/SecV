module sft(
    input wire clk,
    input wire rst,
    input wire [3:0] a,
    output reg [3:0] q
);
    // Internal signal to select mode
    reg mode; // 0 for serial, 1 for parallel

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            q <= 4'b0000;
        end else begin
            if (mode) begin
                // Parallel Data Transfer
                q <= a;
            end else begin
                // Serial Data Transfer
                q <= {q[2:0], a[0]};
            end
        end
    end
endmodule
