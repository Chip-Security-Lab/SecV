module sha256_wrapper(
    input wire clk_i,
    input wire rst_ni,
    input wire rst_3,
    input wire [7:0] wdata,
    input wire [1:0] addr,
    input wire we,
    input wire hashValid,
    output reg startHash,
    output reg [7:0] data0,
    output reg [7:0] data1,
    output reg [7:0] data2,
    output reg [7:0] data3
);

parameter REG_COUNT = 4;

reg [7:0] data_regs [REG_COUNT-1:0];
reg hashValid_r;

always @(posedge clk_i) begin
    if (~(rst_ni && ~rst_3)) begin
        startHash <= 0;
        data_regs <= {REG_COUNT{8'h00}};
    end else if (hashValid && ~hashValid_r) begin
        data_regs <= {REG_COUNT{8'h00}};
    end else if (we) begin
        case (addr)
            2'b00: begin
                data_regs[0] <= wdata;
                startHash <= wdata[0]; 
            end
            2'b01: data_regs[1] <= wdata;
            2'b10: data_regs[2] <= wdata;
            2'b11: data_regs[3] <= wdata;
        endcase
    end

    hashValid_r <= hashValid;
end

assign data0 = data_regs[0];
assign data1 = data_regs[1];
assign data2 = data_regs[2];
assign data3 = data_regs[3];

endmodule
