module sha256_wrapper #(
  parameter REG_COUNT = 4
) (
  input wire clk_i,
  input wire rst_ni,
  input wire rst_3,
  input wire [7:0] wdata,
  input wire [1:0] addr,
  input wire we,
  input wire hashValid,
  output reg startHash,
  output reg [7:0] data0,
  output reg [7:0] data1,
  output reg [7:0] data2,
  output reg [7:0] data3
);

  // Reset behavior
  always @(posedge clk_i or negedge rst_ni or negedge rst_3) begin
    if (!rst_ni || !rst_3) begin
      startHash <= 1'b0;
      data0 <= 8'h0;
      data1 <= 8'h0;
      data2 <= 8'h0;
      data3 <= 8'h0;
    end
    else begin
      // Case 1: Clear only on reset
      // data0 <= 8'h0;
      // data1 <= 8'h0;
      // data2 <= 8'h0;
      // data3 <= 8'h0;

      // Case 2: Clear on reset and hash completion
      if (hashValid) begin
        data0 <= 8'h0;
        data1 <= 8'h0;
        data2 <= 8'h0;
        data3 <= 8'h0;
      end

      if (we) begin
        case (addr)
          2'b00: begin
            data0 <= wdata;
            startHash <= wdata[0];
          end
          2'b01: data1 <= wdata;
          2'b10: data2 <= wdata;
          2'b11: data3 <= wdata;
        endcase
      end
    end
  end

endmodule
