module sha256_wrapper(
    input wire clk_i,
    input wire rst_ni,
    input wire rst_3,
    input wire [7:0] wdata,
    input wire [1:0] addr,
    input wire we,
    input wire hashValid,
    output reg startHash,
    output reg [7:0] data0,
    output reg [7:0] data1,
    output reg [7:0] data2,
    output reg [7:0] data3
);
    
    // Behavioral model for reset and clearing of data registers
    always @(posedge clk_i or negedge rst_ni or negedge rst_3) begin
        if (!rst_ni || !rst_3) begin
            // Reset state
            data0 <= 8'b0;
            data1 <= 8'b0;
            data2 <= 8'b0;
            data3 <= 8'b0;
            startHash <= 1'b0;
        end else begin
            // Normal operation
            if (we) begin
                case (addr)
                    2'b00: begin
                        data0 <= wdata; 
                        startHash <= wdata[0]; // Extract LSB to control hashing
                    end
                    2'b01: data1 <= wdata;
                    2'b10: data2 <= wdata;
                    2'b11: data3 <= wdata;
                endcase
            end
        end
    end

    // Additional behavior for clearing data after hash operation is complete
    reg hashValid_r; // Register to detect rising edge of hashValid
    
    // Track the previous state of hashValid
    always @(posedge clk_i) begin
        hashValid_r <= hashValid;
    end
    
    // Clearing the data registers on rising edge detection of hashValid signal
    always @(posedge clk_i) begin
        if (!hashValid_r && hashValid) begin
            data0 <= 8'b0;
            data1 <= 8'b0;
            data2 <= 8'b0;
            data3 <= 8'b0;
        end
    end

endmodule
