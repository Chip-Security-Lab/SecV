module aes1_wrapper(
    input wire debug_mode_i,
    input wire [KEY_WIDTH-1:0] key_reg0,
    input wire [KEY_WIDTH-1:0] key_reg1,
    output reg [KEY_WIDTH-1:0] core_key0,
    output reg [KEY_WIDTH-1:0] core_key1
);
    always @(*) begin
        if (debug_mode_i) begin
            // Case 1: Partial Clearing
            core_key0 = {KEY_WIDTH{1'b0}};
            core_key1 = key_reg1;
            // Uncomment the following lines for Case 2: Full Clearing
            // core_key0 = {KEY_WIDTH{1'b0}};
            // core_key1 = {KEY_WIDTH{1'b0}};
        end else begin
            core_key0 = key_reg0;
            core_key1 = key_reg1;
        end
    end
endmodule
