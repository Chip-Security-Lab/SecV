module aes1_wrapper #(
    parameter KEY_WIDTH = 8
) (
    input  wire debug_mode_i,
    input  wire [KEY_WIDTH-1:0] key_reg0,
    input  wire [KEY_WIDTH-1:0] key_reg1,
    output reg [KEY_WIDTH-1:0] core_key0,
    output reg [KEY_WIDTH-1:0] core_key1
);

    always @(debug_mode_i, key_reg0, key_reg1) begin
        if (debug_mode_i) begin
            // Case 2: Full Clearing (selected)
            core_key0 <= 0;
            core_key1 <= 0;
        end else begin
            core_key0 <= key_reg0;
            core_key1 <= key_reg1;
        end
    end

endmodule
