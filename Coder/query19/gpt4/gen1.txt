module commit_stage
(
    input wire clk_i,
    input wire rst_ni,
    input wire csr_exception_valid,
    input wire [63:0] csr_exception_cause,
    input wire [1:0] commit_instr_fu,
    input wire [31:0] commit_instr_tval,
    input wire amo_valid_commit_o,
    output wire [63:0] exception_o_cause,
    output wire [31:0] exception_o_tval
);

    // Combinational always block for exception handling logic
    always @(*)
    begin
        // init outputs, here as default no exception case
        exception_o_cause = 64'b0;
        exception_o_tval = 32'b0;

        // exception handling case
        if (csr_exception_valid && csr_exception_cause[63] && !commit_instr_fu[1])
        begin
            // assign cause and tval to outputs
            exception_o_cause = csr_exception_cause;
            exception_o_tval = commit_instr_tval;
        end
    end

endmodule
