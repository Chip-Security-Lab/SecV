module foo_bar(
    input wire clk,
    input wire rst_n,
    input wire [31:0] data_in,
    input wire [31:0] incoming_id,
    input wire [31:0] address,
    output reg [31:0] data_out
);

    reg write_auth;
    reg addr_auth;
    reg [31:0] acl_oh_allowlist = 32'h8312;
    reg [31:0] q;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            acl_oh_allowlist <= 32'h8312;
            q <= 0;
            data_out <= 0;
        end else begin
            write_auth <= (acl_oh_allowlist & incoming_id) ? 1'b1 : 1'b0;
            addr_auth <= ((address == 32'hF00) || (address == 32'h800F00)) ? 1'b1 : 1'b0;

            if (addr_auth && write_auth) begin
                q <= data_in;
            end
            data_out <= q;
        end
    end
endmodule
