module foo_bar(
    input wire [31:0] data_in,
    input wire [31:0] incoming_id,
    input wire [31:0] address,
    input wire clk,
    input wire rst_n,
    output reg [31:0] data_out
);
    // Internal Registers and Wires
    reg [31:0] acl_oh_allowlist;
    reg [31:0] q;
    wire write_auth;
    wire addr_auth;

    // Address Validation Logic - Case 1: Strict Address Validation
    assign addr_auth = (address == 32'hF00);

    // Permission Checks
    assign write_auth = (acl_oh_allowlist & incoming_id) != 0;

    // Reset Behavior
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            acl_oh_allowlist <= 32'h8312;
            q <= 32'h0;
            data_out <= 32'h0;
        end else begin
            if (addr_auth && write_auth) begin
                q <= data_in;
            end
            data_out <= q;
        end
    end
endmodule
