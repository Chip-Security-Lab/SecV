module foo_bar (
    input wire [31:0] data_in,
    input wire [31:0] incoming_id,
    input wire [31:0] address,
    input wire clk,
    input wire rst_n,
    output wire [31:0] data_out
);

    wire write_auth;
    wire addr_auth;
    reg [31:0] acl_oh_allowlist;
    reg [31:0] q;

    assign write_auth = (incoming_id | acl_oh_allowlist) ? 1 : 0;
    assign addr_auth = (address == 32'hF00) ? 1 : 0; // Case 1: Strict Address Validation

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            acl_oh_allowlist <= 32'h8312;
            q <= 0;
        end else if (write_auth && addr_auth) begin
            q <= data_in;
        end
    end

    assign data_out = q;

endmodule
