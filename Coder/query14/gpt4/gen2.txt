module register_write_once_example(
    input wire [15:0] Data_in,
    input wire Clk,
    input wire ip_resetn,
    input wire global_resetn,
    input wire write,
    output reg [15:0] Data_out
);
    reg Write_once_status;

    // Asynchronous reset and synchronous write logic
    always @(posedge Clk or negedge ip_resetn) begin
        if (!ip_resetn || !global_resetn) begin
            Data_out <= 16'h0000;
            Write_once_status <= 1'b0;
        end else if (write && !Write_once_status) begin
            Data_out <= Data_in;
            // Option for Case 1 - Uncomment to use
            // Write_once_status <= 1'b1;

            // Option for Case 2 - Uncomment to use
            // Write_once_status <= Data_in[0];
        end else begin
            Data_out[15:1] <= Data_out[15:1];
            Data_out[0] <= Write_once_status;
        end
    end
endmodule
