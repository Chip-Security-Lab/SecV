module register_write_once_example(
    input wire clk,
    input wire ip_resetn,
    input wire global_resetn,
    input wire write,
    input wire [15:0] Data_in,
    output reg [15:0] Data_out,
    output reg write_once_status
);

    always @(posedge clk or negedge ip_resetn) begin
        if (!ip_resetn) begin
            Data_out <= 16'h0000;
            write_once_status <= 1'b0;
        end else begin
            if (write && !write_once_status) begin
                Data_out[15:1] <= Data_in[15:1];
                write_once_status <= 1'b1;
            end else begin
                Data_out[0] <= Data_in[0];
            end
        end
    end

endmodule
