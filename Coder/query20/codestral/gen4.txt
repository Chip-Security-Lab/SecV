module axi(
  input logic [3:0] access_ctrl_i[NB_SUBORDINATE * NB_MANAGER - 1:0],
  input logic [2:0] priv_lvl_i,
  output logic connectivity_map_o[NB_SUBORDINATE * NB_MANAGER - 1:0]
);

  parameter logic NB_SUBORDINATE = 4, NB_MANAGER = 2;

  // Shared Permissions
  generate
    for (int i = 0; i < NB_SUBORDINATE; i = i + 1) begin
      logic temp;
      assign temp = access_ctrl_i[i * 3 + priv_lvl_i];
      assign connectivity_map_o[i * NB_MANAGER +: NB_MANAGER] = {NB_MANAGER {temp}};
    end
  endgenerate

  //     end

endmodule
