module axi #(
    parameter NB_SUBORDINATE = 4,
    parameter NB_MANAGER = 2
) (
    input wire [NB_SUBORDINATE * 3 * NB_MANAGER - 1 : 0] access_ctrl_i,
    input wire [2 : 0] priv_lvl_i,
    output wire [NB_SUBORDINATE * NB_MANAGER - 1 : 0] connectivity_map_o
);

    genvar i, j;

    generate
        for (i = 0; i < NB_SUBORDINATE; i++) begin
            for (j = 0; j < NB_MANAGER; j++) begin
                // Case 2: Independent Permissions
                assign connectivity_map_o[i * NB_MANAGER + j] = access_ctrl_i[(i * NB_MANAGER + j) * 3 + priv_lvl_i];
            end
        end
    endgenerate

endmodule
