module register_lock(
    input wire clk_i,
    input wire rst_ni,
    input wire jtag_unlock,
    input wire rst_9
);
    reg [5:0] reglk_mem;

    always @(posedge clk_i) begin
        if (~(rst_ni && ~jtag_unlock && ~rst_9)) begin
            for (int j = 0; j < 6; j = j + 1) begin
                reglk_mem[j] <= 'h0; 
            end
        end
    end

    assign reglk_mem_o = reglk_mem;
    output reg [5:0] reglk_mem_o;
endmodule
