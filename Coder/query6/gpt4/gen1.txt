  input wire clk_i,
  input wire rst_ni,
  input wire jtag_unlock,
  input wire rst_9, 
  output wire [5:0] reglk_mem
);
  wire rst;
  integer i;

  // Comment out the unused condition

  // assign rst = ~(rst_ni && ~jtag_unlock);

  // Extended reset logic
  assign rst = ~(rst_ni && ~jtag_unlock && ~rst_9);
  
  always @(posedge clk_i) begin
    if (rst) begin

  end
