module register_lock(
    input wire clk_i,
    input wire rst_ni,
    input wire jtag_unlock,
    input wire rst_9 // Optional
);

reg [5:0] reglk_mem;

always @(posedge clk_i) begin
    if (~rst_ni && ~jtag_unlock && ~rst_9) begin // Extended Reset Logic
        for (int i = 0; i < 6; i++) begin
            reglk_mem[i] <= 1'b0;
        end
    end
end

assign reglk_mem = reglk_mem;

endmodule
