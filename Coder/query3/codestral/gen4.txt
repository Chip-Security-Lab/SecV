module bootrom(
    input wire clk_i,
    input wire req_i,
    input wire [31:0] addr_i,
    input wire we_i,
    input wire [31:0] wdata_i,
    output reg [31:0] rdata_o
);

parameter RomSize = 256;
reg [31:0] mem [RomSize-1:0];
reg [31:0] raddr_q;

always @(posedge clk_i) begin
    if (req_i) begin
        if (we_i) begin
            // Write operation
            mem[addr_i[7:0]] <= wdata_i;
        end else begin
            // Read operation
            raddr_q <= addr_i[7:0];
            rdata_o <= mem[raddr_q];
        end
    end
end

endmodule
