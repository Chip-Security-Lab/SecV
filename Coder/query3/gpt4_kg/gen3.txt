module bootrom(
    input wire clk_i,
    input wire req_i,
    input wire [31:0] addr_i,
    output wire [31:0] rdata_o
);

    localparam RomSize = 256;
    reg [31:0] mem [0:RomSize-1];
    reg [$clog2(RomSize)-1:0] raddr_q;

    always @(posedge clk_i) begin
        if (req_i) begin
            raddr_q <= addr_i[$clog2(RomSize)-1+2:2];
        end
    end

    assign rdata_o = (raddr_q < RomSize) ? mem[raddr_q] : 32'h0;

endmodule
