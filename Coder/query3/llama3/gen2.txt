module bootrom(
    input wire clk_i,
    input wire req_i,
    input wire [31:0] addr_i,
    input wire we_i,
    input wire [31:0] wdata_i,
    output reg [31:0] rdata_o
);

localparam RomSize = 256;

reg [31:0] mem [RomSize-1:0];
reg [31:0] raddr_q;

always @(posedge clk_i) begin
    if (req_i) begin
        if (we_i) begin
            if (addr_i < RomSize) begin
                mem[addr_i] <= wdata_i;
            end
        end else begin
            if (addr_i < RomSize) begin
                raddr_q <= addr_i;
            end
        end
    end
end

always @(posedge clk_i) begin
    if (req_i) begin
        if (raddr_q < RomSize) begin
            rdata_o <= mem[raddr_q];
        end else begin
            rdata_o <= 32'h0;
        end
    end
end

initial begin
    for (int i = 0; i < RomSize; i++) begin
        mem[i] <= 32'h0;
    end
end

endmodule
