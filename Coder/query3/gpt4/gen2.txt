module bootrom #(
    parameter RomSize = 256
)(
    input wire clk_i,
    input wire req_i,
    input wire [31:0] addr_i,
    output reg [31:0] rdata_o
);

    // Memory array
    reg [31:0] mem [0:RomSize-1];
    reg [31:0] raddr_q;

    always @(posedge clk_i) begin
        if (req_i) begin
            raddr_q <= addr_i; // Update read address register
        end
    end

    // Read data from memory
    always @(*) begin
        if (addr_i < RomSize) begin
            rdata_o = mem[raddr_q]; // Output the data at the current address
        end else begin
            rdata_o = 32'h0; // Address out of range
        end
    end

endmodule
