module bootrom(
    input wire clk_i,
    input wire req_i,
    input wire [31:0] addr_i,
    output reg [31:0] rdata_o
);

parameter RomSize = 256;
reg [31:0] mem [0:RomSize-1];
reg [31:0] raddr_q;

always @(posedge clk_i) begin
    if (req_i) begin
        raddr_q <= addr_i;
    end
end

always @(*) begin
    if (raddr_q < RomSize) begin
        rdata_o = mem[raddr_q];
    end else begin
        rdata_o = 32'h0;
    end
end

endmodule
