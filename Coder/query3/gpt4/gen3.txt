module bootrom(
    input wire clk_i,
    input wire req_i,
    input wire [31:0] addr_i,
    output reg [31:0] rdata_o
    );

    parameter RomSize = 256;
    reg [31:0] mem[0:RomSize-1];

    // Using Read-Only mode as no we_i or wdata_i signals provided
    always @(posedge clk_i) begin
        if (req_i) begin
            if (addr_i < RomSize) begin
                rdata_o <= mem[addr_i];
            end else begin
                rdata_o <= 32'h0;
            end
        end
    end

endmodule
