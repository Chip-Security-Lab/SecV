module bootrom(
    input wire clk_i,
    input wire req_i,
    input wire [31:0] addr_i,
    input wire we_i,  // Used only for Read-Write mode
    input wire [31:0] wdata_i,  // Used only for Read-Write mode
    output reg [31:0] rdata_o
);
    parameter RomSize = 256;
    reg [31:0] mem[0:RomSize-1];
    reg [31:0] raddr_q;

    always @(posedge clk_i) begin
        if (req_i) begin
            if (!we_i) begin
                raddr_q <= addr_i;
                if (addr_i < RomSize)
                    rdata_o <= mem[addr_i];
                else
                    rdata_o <= 32'h0;
            end
            else begin
                if (addr_i < RomSize)
                    mem[addr_i] <= wdata_i;
            end
        end
    end

endmodule
