module mod_exp(
    input wire clk,
    input wire rst,
    input wire [WIDTH-1:0] base_in,
    input wire [WIDTH-1:0] exponent_in,
    input wire [WIDTH-1:0] modulus_in,
    output reg [WIDTH-1:0] result_out
);

parameter WIDTH = 32;

reg [WIDTH-1:0] base_reg;
reg [WIDTH-1:0] exponent_reg;
reg [WIDTH-1:0] result_reg;
reg [WIDTH-1:0] mask_reg;
reg [1:0] state;

localparam IDLE = 2'b00;
localparam UPDATE = 2'b01;

always @(posedge clk or posedge rst) begin
    if (rst) begin
        base_reg <= 0;
        exponent_reg <= 0;
        result_reg <= 1;
        mask_reg <= 1;
        state <= IDLE;
    end else begin
        case (state)
            IDLE: begin
                base_reg <= base_in;
                exponent_reg <= exponent_in;
                result_reg <= 1;
                mask_reg <= 1;
                state <= UPDATE;
            end
            UPDATE: begin
                if (exponent_reg != 0) begin
                    if (exponent_reg[0]) begin
                        result_reg <= (result_reg * base_reg) % modulus_in;
                    end else begin
                        // Enhanced handling
                        mask_reg <= (mask_reg * base_reg) % modulus_in;
                    end
                    exponent_reg <= exponent_reg >> 1;
                end else begin
                    result_out <= result_reg;
                    state <= IDLE;
                end
            end
        endcase
    end
end

assign result_out = result_reg;

endmodule
