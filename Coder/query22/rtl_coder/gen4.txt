module mod_exp #(
parameter WIDTH = 32
) (
input wire clk,
input wire rst,
input wire [WIDTH-1:0] base_in,
input wire [WIDTH-1:0] exponent_in,
input wire [WIDTH-1:0] modulus_in,
output reg [WIDTH-1:0] result_out
);

reg [WIDTH-1:0] base_reg;
reg [WIDTH-1:0] exponent_reg;
reg [WIDTH-1:0] result_reg;
reg [WIDTH-1:0] mask_reg;

reg [1:0] state;
parameter IDLE = 2'b00;
parameter UPDATE = 2'b01;

always @(posedge clk) begin
    if (rst) begin
        result_reg <= 1'b1;
        mask_reg <= 1'b1;
        base_reg <= 0;
        exponent_reg <= 0;
        state <= IDLE;
    end else begin
        case (state)
            IDLE: begin
                base_reg <= base_in;
                exponent_reg <= exponent_in;
                result_reg <= 1'b1;
                mask_reg <= 1'b1;
                state <= UPDATE;
            end
            UPDATE: begin
                if (exponent_reg[0] == 1) begin
                    result_reg <= (result_reg * base_reg) % modulus_in;
                end else begin
                    mask_reg <= (mask_reg * base_reg) % modulus_in;
                end
                base_reg <= (base_reg * base_reg) % modulus_in;
                exponent_reg <= exponent_reg >> 1;
                if (exponent_reg == 0) begin
                    state <= IDLE;
                end
            end
        endcase
    end
end

assign result_out = result_reg;

endmodule
