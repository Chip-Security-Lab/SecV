module mod_exp #(parameter WIDTH = 32)(
  input wire clk,
  input wire rst,
  input wire [WIDTH-1:0] base_in,
  input wire [WIDTH-1:0] exponent_in,
  input wire [WIDTH-1:0] modulus_in,
  output wire [WIDTH-1:0] result_out
);
  // Internal Registers
  reg [WIDTH-1:0] base_reg, exponent_reg, result_reg, mask_reg;
  reg [1:0] state;
  
  // States
  localparam IDLE = 2'b00, 
             UPDATE = 2'b01;
  
  always @(posedge clk or posedge rst) begin
    if (rst) begin
      base_reg <= 0;
      exponent_reg <= 0;
      result_reg <= 1;
      mask_reg <= 1;
      state <= IDLE;
    end 
    else begin
      case (state)
        IDLE: begin
          base_reg <= base_in;
          exponent_reg <= exponent_in;
          result_reg <= 1;
          mask_reg <= 1;
          state <= UPDATE;
        end
        UPDATE: begin
          if (exponent_reg == 0) begin
            state <= IDLE;
          end 
          else begin
            if (exponent_reg[0] == 1) 
              result_reg <= (result_reg * base_reg) % modulus_in;
            else 
              mask_reg <= (mask_reg * base_reg) % modulus_in;
            exponent_reg <= exponent_reg >> 1;
            base_reg <= (base_reg * base_reg) % modulus_in;
          end
        end
      endcase
    end
  end
  
  assign result_out = result_reg;
endmodule
