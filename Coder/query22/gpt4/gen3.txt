module mod_exp #(parameter WIDTH = 32) (
  input wire clk,
  input wire rst,
  input wire [WIDTH-1:0] base_in,
  input wire [WIDTH-1:0] exponent_in,
  input wire [WIDTH-1:0] modulus_in,
  output reg [WIDTH-1:0] result_out
);

// Internal registers
reg [WIDTH-1:0] base_reg;
reg [WIDTH-1:0] exponent_reg;
reg [WIDTH-1:0] result_reg;
reg [WIDTH-1:0] mask_reg;
typedef enum { IDLE, UPDATE } State_t;
reg State_t state;

always @(posedge clk or posedge rst) begin
  if (rst) begin
    // Reset behavior
    result_reg <= 1'b1;
    mask_reg <= 1'b1;
    base_reg <= 1'b0;
    exponent_reg <= 1'b0;
    state <= IDLE;
  end else begin
    case (state)
      IDLE: begin
        base_reg <= base_in;
        exponent_reg <= exponent_in;
        state <= UPDATE;
      end
      
      UPDATE: begin
        if (exponent_reg == 0)
          state <= IDLE;
        else if (exponent_reg[0]) begin
          result_reg <= (result_reg * base_reg) % modulus_in;
          base_reg <= (base_reg * base_reg) % modulus_in;
          exponent_reg <= exponent_reg >> 1;
        end else begin
          base_reg <= (base_reg * base_reg) % modulus_in;
          exponent_reg <= exponent_reg >> 1;
        end
      end
    endcase
  end
  result_out <= result_reg;
end

endmodule
