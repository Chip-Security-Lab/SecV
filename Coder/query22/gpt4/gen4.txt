module mod_exp
#(
  parameter WIDTH = 32
)
(
  input wire clk,
  input wire rst,
  input wire [WIDTH-1:0] base_in,
  input wire [WIDTH-1:0] exponent_in,
  input wire [WIDTH-1:0] modulus_in,
  output reg [WIDTH-1:0] result_out
);
  // Internal registers
  reg [WIDTH-1:0] base_reg;
  reg [WIDTH-1:0] exponent_reg;
  reg [WIDTH-1:0] result_reg;
  reg [WIDTH-1:0] mask_reg;
  localparam IDLE = 2'b00, UPDATE = 2'b01;
  reg [1:0] state;

  always @(posedge clk or posedge rst) begin
    if (rst) begin
      base_reg <= 0;
      exponent_reg <= 0;
      result_reg <= 1;
      mask_reg <= 1;
      state <= IDLE;
    end
    else begin
      case (state)
        IDLE: begin
          base_reg <= base_in;
          exponent_reg <= exponent_in;
          state <= UPDATE;
        end
        UPDATE: begin
          if (exponent_reg == 0) state <= IDLE;
          else begin
            if (exponent_reg[0] == 1) result_reg <= base_reg * result_reg % modulus_in;
            base_reg <= base_reg * base_reg % modulus_in;
            exponent_reg <= exponent_reg >> 1;
          end
        end
      endcase
    end
  end

  always @(posedge clk or posedge rst) begin
    if (rst) result_out <= 0;
    else if (state == IDLE) result_out <= result_reg;
  end

endmodule
