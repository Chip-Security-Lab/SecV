module csr_regfile #
(
    parameter WIDTH = 64
)
(
    input  wire                clk,
    input  wire                rst,
    input  wire                csr_we,
    input  wire [WIDTH-1:0]    csr_wdata,
    input  wire [WIDTH-1:0]    mie_q,
    input  wire [WIDTH-1:0]    mideleg_q,
    input  wire [11:0]        csr_addr,
    input  wire [WIDTH-1:0]    utval_q, // Optional
    output reg [WIDTH-1:0]     mie_d
);

always @(posedge clk or posedge rst) begin
    if (rst) begin
        mie_d <= 0;
    end else begin
        if (csr_we) begin
            case (csr_addr)
                12'h104: begin // CSR_SIE
                    // Case 1: With utval_q Influence
                    mie_d <= (mie_q & ~mideleg_q) | (csr_wdata & mideleg_q) | utval_q;
                    // Uncomment the above line and comment the below line to use Case 1
                    // mie_d <= (mie_q & ~mideleg_q) | (csr_wdata & mideleg_q);
                end
                default: begin
                    mie_d <= mie_q;
                end
            endcase
        end else begin
            mie_d <= mie_q;
        end
    end
end

endmodule
