module csr_regfile #(parameter WIDTH = 64) (
  input wire clk,
  input wire rst,
  input wire csr_we,
  input wire [WIDTH-1:0] csr_wdata,
  input wire [WIDTH-1:0] mie_q,
  input wire [WIDTH-1:0] mideleg_q,
  input wire [11:0] csr_addr,
  input wire [WIDTH-1:0] utval_q,
  output reg [WIDTH-1:0] mie_d
);

  always @(posedge clk or posedge rst) begin
    if (rst)
      mie_d <= {WIDTH{1'b0}};
    else if (csr_we) begin
      case (csr_addr)
        12'h104:
          // Select either case 1 or case 2 logic for mie_d update
          // Case 1: with utval_q influence
          // mie_d <= (mie_q & ~mideleg_q) | (csr_wdata & mideleg_q) | utval_q;
  
          // Case 2: without utval_q influence
          mie_d <= (mie_q & ~mideleg_q) | (csr_wdata & mideleg_q);
        default: mie_d <= mie_q;
      endcase
    end
  end
endmodule
