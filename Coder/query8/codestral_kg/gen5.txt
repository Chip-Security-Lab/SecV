module fsm(
    input  wire [3:0] register_address,
    output reg        gpio_out
);
    always @(*) begin
        gpio_out = 0;
        case (register_address)
            4'h0: gpio_out = 0;
            4'h1: gpio_out = 1;
            4'h2: gpio_out = 1;
            4'h3: gpio_out = 0;
            4'h4: gpio_out = 1;
            4'h5: gpio_out = 0;
            4'h6: gpio_out = 0;
            4'h7: gpio_out = 1;
            4'h8: gpio_out = 0;
            4'h9: gpio_out = 1;
            4'ha: gpio_out = 0;
            4'hb: gpio_out = 0;
            4'hc: gpio_out = 1;
            4'hd: gpio_out = 1;
            4'he: gpio_out = 1;
            4'hf: gpio_out = 0;
            default: gpio_out = gpio_out;
        endcase
    end
endmodule
