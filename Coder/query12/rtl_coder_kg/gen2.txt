module aes0_wrapper(
  input wire clk_i,
  input wire rst_ni,
  input wire rst_1,
  input wire en,
  input wire we,
  input wire [ADDR_WIDTH-1:0] address,
  input wire [DATA_WIDTH-1:0] wdata,
  input wire ct_valid,
  input wire [3:0] reglk_ctrl_i,
  output reg [DATA_WIDTH-1:0] p_c0,
  output reg [DATA_WIDTH-1:0] p_c1,
  output reg [DATA_WIDTH-1:0] p_c2,
  output reg [DATA_WIDTH-1:0] p_c3,
  output reg start
);

  // Register clearing logic
  always @(posedge clk_i or negedge rst_ni or negedge rst_1) begin
    if (~(rst_ni && ~rst_1)) begin
      start <= 0;
      p_c0 <= 0;
      p_c1 <= 0;
      p_c2 <= 0;
      p_c3 <= 0;
    end else if (ct_valid) begin
      p_c0 <= 0;
      p_c1 <= 0;
      p_c2 <= 0;
      p_c3 <= 0;
    end
  end

  // Register update logic
  always @(posedge clk_i) begin
    if (en && we) begin
      case (address[8:3])
        0: begin
          if (reglk_ctrl_i[1]) begin
            start <= start;
          end else begin
            start <= wdata[0];
          end
        end
        1: begin
          if (reglk_ctrl_i[2]) begin
            p_c3 <= p_c3;
          end else begin
            p_c3 <= wdata;
          end
        end
        2: begin
          if (reglk_ctrl_i[3]) begin
            p_c2 <= p_c2;
          end else begin
            p_c2 <= wdata;
          end
        end
        3: begin
          if (reglk_ctrl_i[4]) begin
            p_c1 <= p_c1;
          end else begin
            p_c1 <= wdata;
          end
        end
        4: begin
          if (reglk_ctrl_i[5]) begin
            p_c0 <= p_c0;
          end else begin
            p_c0 <= wdata;
          end
        end
      endcase
    end
  end

endmodule
