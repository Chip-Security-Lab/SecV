module aes0_wrapper #(
parameter ADDR_WIDTH = 8,
parameter DATA_WIDTH = 32
) (
input wire clk_i,
input wire rst_ni,
input wire rst_1,
input wire en,
input wire we,
input wire [ADDR_WIDTH-1:0] address,
input wire [DATA_WIDTH-1:0] wdata,
input wire ct_valid,
input wire [3:0] reglk_ctrl_i,
output reg [DATA_WIDTH-1:0] p_c0,
output reg [DATA_WIDTH-1:0] p_c1,
output reg [DATA_WIDTH-1:0] p_c2,
output reg [DATA_WIDTH-1:0] p_c3,
output reg start
);

// Register clearing logic
reg [DATA_WIDTH-1:0] p_c0_reg, p_c1_reg, p_c2_reg, p_c3_reg;
reg start_reg;

// Reset behavior
always @(posedge clk_i or negedge rst_ni or negedge rst_1) begin
    if (!rst_ni || !rst_1) begin
        p_c0_reg <= 0;
        p_c1_reg <= 0;
        p_c2_reg <= 0;
        p_c3_reg <= 0;
        start_reg <= 0;
    end else begin
        // Register clearing logic
        case (1'b1)
            // Clear only on reset
            reglk_ctrl_i[0]: p_c0_reg <= p_c0_reg;
            reglk_ctrl_i[1]: p_c1_reg <= p_c1_reg;
            reglk_ctrl_i[2]: p_c2_reg <= p_c2_reg;
            reglk_ctrl_i[3]: p_c3_reg <= p_c3_reg;
            // Clear on reset and cryptographic completion
            ct_valid: begin
                p_c0_reg <= 0;
                p_c1_reg <= 0;
                p_c2_reg <= 0;
                p_c3_reg <= 0;
            end
        endcase

        // Register update logic
        if (en && we) begin
            case (address[ADDR_WIDTH-1:3])
                0: start_reg <= wdata[0] & reglk_ctrl_i[1];
                1: p_c3_reg <= wdata;
                2: p_c2_reg <= wdata;
                3: p_c1_reg <= wdata;
                4: p_c0_reg <= wdata;
            endcase
        end
    end
end

// Output assignments
assign p_c0 = p_c0_reg;
assign p_c1 = p_c1_reg;
assign p_c2 = p_c2_reg;
assign p_c3 = p_c3_reg;
assign start = start_reg;

endmodule
