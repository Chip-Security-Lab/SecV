module aes0_wrapper #(
    parameter ADDR_WIDTH = 8,
    parameter DATA_WIDTH = 32
) (
    input  wire clk_i,
    input  wire rst_ni,
    input  wire rst_1,
    input  wire en,
    input  wire we,
    input  wire [ADDR_WIDTH-1:0] address,
    input  wire [DATA_WIDTH-1:0] wdata,
    input  wire ct_valid,
    input  wire [3:0] reglk_ctrl_i,
    output reg [DATA_WIDTH-1:0] p_c0,
    output reg [DATA_WIDTH-1:0] p_c1,
    output reg [DATA_WIDTH-1:0] p_c2,
    output reg [DATA_WIDTH-1:0] p_c3,
    output reg start
);

    // Case 2: Clear on Reset and Cryptographic Completion
    always @(posedge clk_i) begin
        if (~rst_ni || ~rst_1) begin
            p_c0 <= 0;
            p_c1 <= 0;
            p_c2 <= 0;
            p_c3 <= 0;
            start <= 0;
        end else if (ct_valid) begin
            p_c0 <= 0;
            p_c1 <= 0;
            p_c2 <= 0;
            p_c3 <= 0;
        end else if (en && we) begin
            case (address[8:3])
                0: begin
                    if (~reglk_ctrl_i[1])
                        start <= wdata[0];
                end
                1: begin
                    if (~reglk_ctrl_i[0])
                        p_c3 <= wdata;
                end
                2: begin
                    if (~reglk_ctrl_i[0])
                        p_c2 <= wdata;
                end
                3: begin
                    if (~reglk_ctrl_i[0])
                        p_c1 <= wdata;
                end
                4: begin
                    if (~reglk_ctrl_i[0])
                        p_c0 <= wdata;
                end
            endcase
        end
    end

endmodule
