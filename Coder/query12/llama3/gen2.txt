module aes0_wrapper #(
    parameter ADDR_WIDTH = 8,
    parameter DATA_WIDTH = 32
) (
    input  wire clk_i,
    input  wire rst_ni,
    input  wire rst_1,
    input  wire en,
    input  wire we,
    input  wire [ADDR_WIDTH-1:0] address,
    input  wire [DATA_WIDTH-1:0] wdata,
    input  wire ct_valid,
    input  wire [3:0] reglk_ctrl_i,
    output reg [DATA_WIDTH-1:0] p_c0,
    output reg [DATA_WIDTH-1:0] p_c1,
    output reg [DATA_WIDTH-1:0] p_c2,
    output reg [DATA_WIDTH-1:0] p_c3,
    output reg start
);

    // Register declarations
    reg [DATA_WIDTH-1:0] reg_c0;
    reg [DATA_WIDTH-1:0] reg_c1;
    reg [DATA_WIDTH-1:0] reg_c2;
    reg [DATA_WIDTH-1:0] reg_c3;
    reg reg_start;

    // Reset logic
    always @(posedge clk_i or negedge rst_ni or negedge rst_1) begin
        if (~rst_ni || ~rst_1) begin
            reg_c0 <= 0;
            reg_c1 <= 0;
            reg_c2 <= 0;
            reg_c3 <= 0;
            reg_start <= 0;
        end
    end

    // Case 2: Clear on Reset and Cryptographic Completion
    // Uncomment the following lines for Case 2
    // always @(posedge clk_i) begin
    //     if (ct_valid) begin
    //         reg_c0 <= 0;
    //         reg_c1 <= 0;
    //         reg_c2 <= 0;
    //         reg_c3 <= 0;
    //     end
    // end

    // Register update logic
    always @(posedge clk_i) begin
        if (en && we) begin
            case (address[ADDR_WIDTH-1:3])
                0: begin
                    if (~reglk_ctrl_i[1]) begin
                        reg_start <= wdata[0];
                    end
                end
                1: begin
                    if (~reglk_ctrl_i[0]) begin
                        reg_c3 <= wdata;
                    end
                end
                2: begin
                    if (~reglk_ctrl_i[0]) begin
                        reg_c2 <= wdata;
                    end
                end
                3: begin
                    if (~reglk_ctrl_i[0]) begin
                        reg_c1 <= wdata;
                    end
                end
                4: begin
                    if (~reglk_ctrl_i[0]) begin
                        reg_c0 <= wdata;
                    end
                end
            endcase
        end
    end

    // Output assignments
    assign p_c0 = reg_c0;
    assign p_c1 = reg_c1;
    assign p_c2 = reg_c2;
    assign p_c3 = reg_c3;
    assign start = reg_start;

endmodule
