module aes0_wrapper #(
    parameter ADDR_WIDTH = 8,
    parameter DATA_WIDTH = 32
)(
    input wire clk_i,
    input wire rst_ni,
    input wire rst_1,
    input wire en,
    input wire we,
    input wire [ADDR_WIDTH-1:0] address,
    input wire [DATA_WIDTH-1:0] wdata,
    input wire ct_valid,
    input wire [3:0] reglk_ctrl_i,
    output reg [DATA_WIDTH-1:0] p_c0,
    output reg [DATA_WIDTH-1:0] p_c1,
    output reg [DATA_WIDTH-1:0] p_c2,
    output reg [DATA_WIDTH-1:0] p_c3,
    output reg start
);

    // Synchronous logic for handling resets, writes, and cryptographic completion
    always @(posedge clk_i) begin
        if (~(rst_ni && ~rst_1)) begin
            // Reset all outputs to 0
            p_c0 <= 0;
            p_c1 <= 0;
            p_c2 <= 0;
            p_c3 <= 0;
            start <= 0;
        end else if (ct_valid) begin
            // Clear cryptographic data on operation completion
            p_c0 <= 0;
            p_c1 <= 0;
            p_c2 <= 0;
            p_c3 <= 0;
        end else if (en && we) begin
            // Write data to registers based on the address and lock status
            case (address[8:3])
                0: start <= reglk_ctrl_i[1] ? start : wdata[0];
                1: p_c3 <= reglk_ctrl_i[3] ? p_c3 : wdata;
                2: p_c2 <= reglk_ctrl_i[2] ? p_c2 : wdata;
                3: p_c1 <= reglk_ctrl_i[1] ? p_c1 : wdata;
                4: p_c0 <= reglk_ctrl_i[0] ? p_c0 : wdata;
            endcase
        end
    end

endmodule
