module aes0_wrapper #(
    parameter ADDR_WIDTH = 8,
    parameter DATA_WIDTH = 32
)(
    input wire clk_i,
    input wire rst_ni,
    input wire rst_1,
    input wire en,
    input wire we,
    input wire [ADDR_WIDTH-1:0] address,
    input wire [DATA_WIDTH-1:0] wdata,
    input wire ct_valid,
    input wire [3:0] reglk_ctrl_i,
    output reg [DATA_WIDTH-1:0] p_c0,
    output reg [DATA_WIDTH-1:0] p_c1,
    output reg [DATA_WIDTH-1:0] p_c2,
    output reg [DATA_WIDTH-1:0] p_c3,
    output reg start
);

    // Reset or clear conditions
    wire reset_condition = !rst_ni || !rst_1;
    wire clear_on_completion = ct_valid;

    always @(posedge clk_i) begin
        if (reset_condition) begin
            p_c0 <= 32'd0;
            p_c1 <= 32'd0;
            p_c2 <= 32'd0;
            p_c3 <= 32'd0;
            start <= 1'b0;
        end
        else if (clear_on_completion) begin
            p_c0 <= 32'd0;
            p_c1 <= 32'd0;
            p_c2 <= 32'd0;
            p_c3 <= 32'd0;
        end
        else if (en && we) begin
            case (address[8:3])
                0: if (!reglk_ctrl_i[1]) start <= wdata[0];
                1: if (!reglk_ctrl_i[0]) p_c3 <= wdata;
                2: if (!reglk_ctrl_i[1]) p_c2 <= wdata;
                3: if (!reglk_ctrl_i[2]) p_c1 <= wdata;
                4: if (!reglk_ctrl_i[3]) p_c0 <= wdata;
            endcase
        end
    end

endmodule
