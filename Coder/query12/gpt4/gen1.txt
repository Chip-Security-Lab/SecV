module aes0_wrapper #(
    parameter ADDR_WIDTH = 8,
    parameter DATA_WIDTH = 32
)(
    input wire clk_i,
    input wire rst_ni,
    input wire rst_1,
    input wire en,
    input wire we,
    input wire [ADDR_WIDTH-1:0] address,
    input wire [DATA_WIDTH-1:0] wdata,
    input wire ct_valid,
    input wire [3:0] reglk_ctrl_i,
    output reg [DATA_WIDTH-1:0] p_c0,
    output reg [DATA_WIDTH-1:0] p_c1,
    output reg [DATA_WIDTH-1:0] p_c2,
    output reg [DATA_WIDTH-1:0] p_c3,
    output reg start
);

always @(posedge clk_i) begin
    if (!rst_ni || !rst_1) begin
        p_c0 <= 32'b0;
        p_c1 <= 32'b0;
        p_c2 <= 32'b0;
        p_c3 <= 32'b0;
        start <= 1'b0;
    end else if (ct_valid) begin
        // Clearing on Cryptographic Completion
        p_c0 <= 32'b0;
        p_c1 <= 32'b0;
        p_c2 <= 32'b0;
        p_c3 <= 32'b0;
    end else if (en && we) begin
        case (address[2:0])
            3'b000: if (!reglk_ctrl_i[1]) start <= wdata[0];
            3'b001: if (!reglk_ctrl_i[0]) p_c3 <= wdata;
            3'b010: if (!reglk_ctrl_i[0]) p_c2 <= wdata;
            3'b011: if (!reglk_ctrl_i[0]) p_c1 <= wdata;
            3'b100: if (!reglk_ctrl_i[0]) p_c0 <= wdata;
            default: ;
        endcase
    end
end

endmodule
