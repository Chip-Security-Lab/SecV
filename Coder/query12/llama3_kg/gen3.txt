module aes0_wrapper #(
    parameter ADDR_WIDTH = 8,
    parameter DATA_WIDTH = 32
) (
    input wire clk_i,
    input wire rst_ni,
    input wire rst_1,
    input wire en,
    input wire we,
    input wire [ADDR_WIDTH-1:0] address,
    input wire [DATA_WIDTH-1:0] wdata,
    input wire ct_valid,
    input wire [3:0] reglk_ctrl_i,
    output reg [DATA_WIDTH-1:0] p_c0,
    output reg [DATA_WIDTH-1:0] p_c1,
    output reg [DATA_WIDTH-1:0] p_c2,
    output reg [DATA_WIDTH-1:0] p_c3,
    output reg start
);

    reg [DATA_WIDTH-1:0] p_c [3:0];

    always @(posedge clk_i) begin
        if (~(rst_ni && ~rst_1)) begin
            p_c[0] <= 0;
            p_c[1] <= 0;
            p_c[2] <= 0;
            p_c[3] <= 0;
            start <= 0;
        end else if (en && we) begin
            case (address[ADDR_WIDTH-1:3])
                4: if (~reglk_ctrl_i[0]) p_c[0] <= wdata;
                3: if (~reglk_ctrl_i[1]) p_c[1] <= wdata;
                2: if (~reglk_ctrl_i[2]) p_c[2] <= wdata;
                1: if (~reglk_ctrl_i[3]) p_c[3] <= wdata;
                0: if (~reglk_ctrl_i[1]) start <= wdata[0];
                default: ;
            endcase
        end else if (ct_valid) begin
            p_c[0] <= 0;
            p_c[1] <= 0;
            p_c[2] <= 0;
            p_c[3] <= 0;
        end
    end

    assign p_c0 = p_c[0];
    assign p_c1 = p_c[1];
    assign p_c2 = p_c[2];
    assign p_c3 = p_c[3];

endmodule
