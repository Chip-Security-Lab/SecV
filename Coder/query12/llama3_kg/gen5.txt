module aes0_wrapper #(
    parameter ADDR_WIDTH = 8,
    parameter DATA_WIDTH = 32
) (
    input wire clk_i,
    input wire rst_ni,
    input wire rst_1,
    input wire en,
    input wire we,
    input wire [ADDR_WIDTH-1:0] address,
    input wire [DATA_WIDTH-1:0] wdata,
    input wire ct_valid,
    input wire [3:0] reglk_ctrl_i,
    output reg [DATA_WIDTH-1:0] p_c0,
    output reg [DATA_WIDTH-1:0] p_c1,
    output reg [DATA_WIDTH-1:0] p_c2,
    output reg [DATA_WIDTH-1:0] p_c3,
    output reg start
);

    reg [DATA_WIDTH-1:0] p_c_reg [0:3];

    always @(posedge clk_i) begin
        if (~rst_ni || ~rst_1) begin
            p_c_reg <= '{default: 0};
            start <= 0;
        end else if (ct_valid) begin
            // Case 2: Clear on Reset and Cryptographic Completion
            p_c_reg <= '{default: 0};
        end else if (en && we) begin
            case (address[ADDR_WIDTH-1:3])
                0: if (~reglk_ctrl_i[1]) start <= wdata[0];
                1: if (~reglk_ctrl_i[0]) p_c_reg[3] <= wdata;
                2: if (~reglk_ctrl_i[0]) p_c_reg[2] <= wdata;
                3: if (~reglk_ctrl_i[0]) p_c_reg[1] <= wdata;
                4: if (~reglk_ctrl_i[0]) p_c_reg[0] <= wdata;
            endcase
        end
    end

    assign p_c0 = p_c_reg[0];
    assign p_c1 = p_c_reg[1];
    assign p_c2 = p_c_reg[2];
    assign p_c3 = p_c_reg[3];

endmodule
