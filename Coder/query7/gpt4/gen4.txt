module fsm_1(
    input wire [2:0] user_input,
    input wire clk,
    input wire rst_n,
    output reg [2:0] out
);
    // FSM State Register
    reg [1:0] state;
    
    // Always block for state transitions
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) state <= 2'h0;    // Reset Condition
        else case (user_input)
            3'h3: state <= 2'h3;      // Transition to state 3
            3'h4: state <= 2'h2;      // Transition to state 2
            3'h5: state <= 2'h1;      // Transition to state 1
            default: state <= state;  // Maintain current state otherwise
        endcase
    end

    // Always block for combinational output logic
    always @(*) begin
        out = {1'b1, state};          // MSB is always '1'. State is LSBs.
    end

endmodule
