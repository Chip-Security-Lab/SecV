module fsm_1(
    input wire clk,
    input wire rst_n,
    input wire [2:0] user_input,
    output reg [2:0] out
);
    reg [1:0] current_state;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            current_state <= 2'h0;
        else begin
            case (user_input)
                3'h3: current_state <= 2'h3;
                3'h4: current_state <= 2'h2;
                3'h5: current_state <= 2'h1;
                default: current_state <= current_state;
            endcase
        end
    end

    always @(*) begin
        out[1:0] = current_state;
        out[2] = 1'b1;
    end
endmodule
