module fsm_1(
    input wire [2:0] user_input,
    input wire clk,
    input wire rst_n,
    output wire [2:0] out
);
    // FSM state declaration
    reg [1:0] curr_state, next_state;

    // State transition logic
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            curr_state <= 2'h0;
        end else begin
            curr_state <= next_state;
        end
    end

    // Next state logic
    always @* begin
        // Default next state
        next_state = curr_state;

        // Conditional transitions
        case (user_input)
            3'h3: next_state = 2'h3;
            3'h4: next_state = 2'h2;
            3'h5: next_state = 2'h1;
        endcase
    end

    // Output logic
    assign out = {1'b1, curr_state};
endmodule
