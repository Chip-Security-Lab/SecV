module fsm_1(
    input wire [2:0] user_input,
    input wire clk,
    input wire rst_n,
    output reg [2:0] out
);

reg [1:0] state;
parameter S0 = 2'h0, S1 = 2'h1, S2 = 2'h2, S3 = 2'h3;

always @(posedge clk or negedge rst_n) begin
    if (~rst_n) begin
        state <= S0;
    end else begin
        case (user_input)
            3'h3: state <= S3;
            3'h4: state <= S2;
            3'h5: state <= S1;
            default: state <= S0;
        endcase
    end
end

always @* begin
    out = {1'b1, state};
end

endmodule
