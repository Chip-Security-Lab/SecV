module fsm_1(
    input wire [2:0] user_input,
    input wire clk,
    input wire rst_n,
    output reg [2:0] out
);

    reg [1:0] state, next_state;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= 2'h0;
        end else begin
            state <= next_state;
        end
    end

    always @* begin
        case (user_input)
            3'h3: next_state = 2'h3;
            3'h4: next_state = 2'h2;
            3'h5: next_state = 2'h1;
            default: next_state = state;
        endcase
    end

    always @* begin
        out[1:0] = state;
        out[2] = 1;
    end

endmodule
