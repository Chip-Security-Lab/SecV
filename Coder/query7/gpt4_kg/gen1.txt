module fsm_1(
    input wire [2:0] user_input,
    input wire clk,
    input wire rst_n,
    output reg [2:0] out
);

// State encoding
localparam S0 = 2'h0;
localparam S1 = 2'h1;
localparam S2 = 2'h2;
localparam S3 = 2'h3;

// Internal State
reg [1:0] state;

// Combinational logic for output
always @(*) begin
    out = {1'b1, state};
end

// Sequential logic for state transition
always @(posedge clk or negedge rst_n) begin
    if (!rst_n)
        state <= S0; // Reset state
    else begin
        case (user_input)
            3'h3: state <= S3;
            3'h4: state <= S2;
            3'h5: state <= S1;
            default: state <= state; // No change
        endcase
    end
end

endmodule
