module fsm_1(
  input wire [2:0] user_input,
  input wire clk,
  input wire rst_n,
  output reg [2:0] out
);

  reg [1:0] state;

  always @(posedge clk or negedge rst_n) begin
    if (~rst_n) begin
      state <= 2'h0;
      out <= 3'b100;
    end else begin
      case (state)
        2'h0: begin
          if (user_input == 3'h3) begin
            state <= 2'h3;
            out <= 3'b111;
          end else if (user_input == 3'h4) begin
            state <= 2'h2;
            out <= 3'b110;
          end else if (user_input == 3'h5) begin
            state <= 2'h1;
            out <= 3'b101;
          end else begin
            state <= 2'h0;
            out <= 3'b100;
          end
        end
        2'h1: begin
          if (user_input == 3'h3) begin
            state <= 2'h3;
            out <= 3'b111;
          end else if (user_input == 3'h4) begin
            state <= 2'h2;
            out <= 3'b110;
          end else if (user_input == 3'h5) begin
            state <= 2'h1;
            out <= 3'b101;
          end else begin
            state <= 2'h0;
            out <= 3'b100;
          end
        end
        2'h2: begin
          if (user_input == 3'h3) begin
            state <= 2'h3;
            out <= 3'b111;
          end else if (user_input == 3'h4) begin
            state <= 2'h2;
            out <= 3'b110;
          end else if (user_input == 3'h5) begin
            state <= 2'h1;
            out <= 3'b101;
          end else begin
            state <= 2'h0;
            out <= 3'b100;
          end
        end
        2'h3: begin
          if (user_input == 3'h3) begin
            state <= 2'h3;
            out <= 3'b111;
          end else if (user_input == 3'h4) begin
            state <= 2'h2;
            out <= 3'b110;
          end else if (user_input == 3'h5) begin
            state <= 2'h1;
            out <= 3'b101;
          end else begin
            state <= 2'h0;
            out <= 3'b100;
          end
        end
      endcase
    end
  end
