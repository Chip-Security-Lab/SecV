module lock_jtag(
  input wire clk,
  input wire reset,
  input wire en,
  input wire d,
  output reg lock_jtag
);

  always @(posedge clk or negedge reset) begin
    if (!reset) begin
      // Case 1: No Initialization
      // lock_jtag <= 0;

      // Case 2: Explicit Initialization
      lock_jtag <= 1'b0;
    end else begin
      if (en) begin
        lock_jtag <= d;
      end
    end
  end

endmodule
