module jtag_auth_good(
    input wire clk_i,
    input wire rst_ni,
    input wire [31:0] pass_hash,
    input wire [31:0] exp_hash,
    input wire hashValid,
    output reg pass_check,
    output reg [1:0] state_q,
    output reg [1:0] state_d,
    output reg [2:0] miss_pass_check_cnt_q
);

    parameter LOCK_THRESH = 3;

    always @(posedge clk_i or negedge rst_ni) begin
        if (~rst_ni) begin
            state_q <= 2'b00; // Idle
            pass_check <= 1'b0;
            miss_pass_check_cnt_q <= 3'b000;
        end else begin
            state_q <= state_d;
            case (state_q)
                2'b00: begin // Idle
                    if (hashValid) begin
                        state_d <= 2'b01; // PassChkValid
                    end else begin
                        state_d <= 2'b00; // Idle
                    end
                end
                2'b01: begin // PassChkValid
                    if (pass_hash == exp_hash) begin
                        pass_check <= 1'b1;
                        state_d <= 2'b00; // Idle
                        miss_pass_check_cnt_q <= 3'b000;
                    end else begin
                        pass_check <= 1'b0;
                        state_d <= 2'b10; // Lock (if LOCK_THRESH is exceeded)
                        miss_pass_check_cnt_q <= miss_pass_check_cnt_q + 1'b1;
                        if (miss_pass_check_cnt_q == LOCK_THRESH) begin
                            state_d <= 2'b10; // Lock
                        end
                    end
                end
                2'b10: begin // Lock
                    pass_check <= 1'b0;
                    state_d <= 2'b10; // Lock
                end
            endcase
        end
    end

endmodule
