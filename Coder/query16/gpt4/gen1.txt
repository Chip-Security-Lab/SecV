module jtag_auth_good(
    input wire clk_i,
    input wire rst_ni,
    input wire [31:0] pass_hash,
    input wire [31:0] exp_hash,
    input wire hashValid,
    output wire pass_check,
    output reg [1:0] state_q,
    output reg [1:0] state_d,
    output reg [2:0] miss_pass_check_cnt_q
);

    // Define the state codes
    localparam [1:0] 
        IDLE = 2'b00,
        PASS_CHK_VALID = 2'b01,
        LOCK = 2'b10;

    // Max threshold for attempts
    localparam MAX_ATTEMPTS = 3'b011;

    // State transitional and combinatorial logic
    always @(posedge clk_i) begin
        if (!rst_ni) begin
            state_q <= IDLE;
            miss_pass_check_cnt_q <= 0;
        end else begin
            state_q <= state_d;
            if (state_q == PASS_CHK_VALID && pass_hash == exp_hash)
                miss_pass_check_cnt_q <= 0;
            else if (state_q == PASS_CHK_VALID && pass_hash != exp_hash)
                miss_pass_check_cnt_q <= miss_pass_check_cnt_q + 1;
        end
    end

    // Next state logic
    always @(*) begin
        state_d = state_q; // Default to current state
        case (state_q)
            IDLE: begin
                if (hashValid)
                    state_d = PASS_CHK_VALID;
            end
            PASS_CHK_VALID: begin
                if (miss_pass_check_cnt_q >= MAX_ATTEMPTS)
                    state_d = LOCK;
                else
                    state_d = IDLE;
            end
            LOCK: begin
                // Maintain lock state unless reset occurs
                state_d = LOCK;
            end
            default: state_d = IDLE;
        endcase
    end

    // Output logic
    assign pass_check = (state_q == PASS_CHK_VALID && pass_hash == exp_hash) ? 1'b1 : 1'b0;

endmodule
