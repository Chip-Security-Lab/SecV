module jtag_auth_good(
    input wire clk_i,
    input wire rst_ni,
    input wire [31:0] pass_hash,
    input wire [31:0] exp_hash,
    input wire hashValid,
    output reg pass_check,
    output reg [2:0] state_q,
    output reg [2:0] state_d,
    output reg [2:0] miss_pass_check_cnt_q
);

    // FSM state encoding
    localparam [2:0] 
        IDLE = 3'b000,
        PASS_CHK_VALID = 3'b001,
        LOCK = 3'b010;

    // Maximum password attempt threshold
    localparam MAX_ATTEMPTS = 3;
    
    // FSM state register (Sequential logic)
    always @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni) begin
            state_q <= IDLE;
            miss_pass_check_cnt_q <= 0;
            pass_check <= 0;
        end else begin
            state_q <= state_d;
            if (state_d == PASS_CHK_VALID && pass_hash == exp_hash) begin
                miss_pass_check_cnt_q <= 0;
            end else if (state_d == PASS_CHK_VALID && pass_hash != exp_hash) begin
                miss_pass_check_cnt_q <= miss_pass_check_cnt_q + 1;
            end
        end
    end

    // FSM next state logic (Combinatorial logic)
    always @(*) begin
        state_d = state_q; // Default state is the current state
        pass_check = 0;
        
        case (state_q)
            IDLE: begin
                if (hashValid) state_d = PASS_CHK_VALID;
            end
            PASS_CHK_VALID: begin
                if (pass_hash == exp_hash) begin
                    state_d = IDLE;
                    pass_check = 1;
                end else begin
                    if (miss_pass_check_cnt_q + 1 >= MAX_ATTEMPTS) begin
                        state_d = LOCK;
                    end else begin
                        state_d = IDLE;
                    end
                end
            end
            LOCK: begin
                // In LOCK state, no further checking; remain in LOCK
                pass_check = 0;
            end
            default: begin
                state_d = IDLE;
            end
        endcase
    end

endmodule
