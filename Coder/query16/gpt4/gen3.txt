module jtag_auth_good(
    input wire clk_i,
    input wire rst_ni,
    input wire [31:0] pass_hash,
    input wire [31:0] exp_hash,
    input wire hashValid,
    output reg pass_check,
    output reg [1:0] state_q,
    output reg [1:0] state_d,
    output reg [2:0] miss_pass_check_cnt_q
);

    // FSM state encoding
    localparam IDLE = 2'b00,
               PASS_CHK_VALID = 2'b01,
               LOCK = 2'b10;

    // Threshold for failed attempts
    localparam MAX_FAILED_ATTEMPTS = 3;
    
    always @(posedge clk_i or negedge rst_ni) begin
        if (!rst_ni) begin
            state_q <= IDLE;
            pass_check <= 1'b0;
            miss_pass_check_cnt_q <= 0;
        end else begin
            state_q <= state_d;
            
            case(state_q)
                IDLE: begin
                    pass_check <= 1'b0;
                    if (hashValid) begin
                        state_d <= PASS_CHK_VALID;
                    end else begin
                        state_d <= IDLE;
                    end
                end
                PASS_CHK_VALID: begin
                    if (pass_hash == exp_hash) begin
                        pass_check <= 1'b1;
                        miss_pass_check_cnt_q <= 0;
                        state_d <= IDLE;
                    end else begin
                        pass_check <= 1'b0;
                        if (miss_pass_check_cnt_q >= MAX_FAILED_ATTEMPTS - 1) begin
                            state_d <= LOCK;
                        end else begin
                            miss_pass_check_cnt_q <= miss_pass_check_cnt_q + 1;
                            state_d <= IDLE;
                        end
                    end
                end
                LOCK: begin
                    pass_check <= 1'b0;
                    // Remain in LOCK state regardless of hashValid and pass_hash
                    state_d <= LOCK;
                end
                default: state_d <= IDLE;
            endcase
        end
    end

endmodule
