module jtag_auth_good(
    input wire clk_i,
    input wire rst_ni,
    input wire [31:0] pass_hash,
    input wire [31:0] exp_hash,
    input wire hashValid,
    output wire pass_check,
    output wire [1:0] state_q,
    output wire [1:0] state_d,
    output wire [1:0] miss_pass_check_cnt_q
);

localparam Idle          = 2'b00;
localparam PassChkValid  = 2'b01;
localparam Lock          = 2'b10;

reg [1:0] miss_pass_check_cnt_d;

always @(posedge clk_i or negedge rst_ni) begin
    if (~rst_ni) begin
        state_q <= Idle;
        pass_check <= 1'b0;
        miss_pass_check_cnt_q <= 2'b00;
    end else begin
        state_q <= state_d;
        miss_pass_check_cnt_q <= miss_pass_check_cnt_d;
    end
end

always @(*) begin
    case (state_q)
        PassChkValid: begin
            if (hashValid) begin
                if (exp_hash == pass_hash) begin
                    pass_check = 1'b1;
                    miss_pass_check_cnt_d = 2'b00;
                end else begin
                    pass_check = 1'b0;
                    miss_pass_check_cnt_d = miss_pass_check_cnt_q + 1;
                    if (miss_pass_check_cnt_q == 2'b11) begin
                        state_d = Lock;
                    end else begin
                        state_d = Idle;
                    end
                end
            end else begin
                state_d = PassChkValid;
            end
        end
        Lock: begin
            pass_check = 1'b0;
            state_d = Lock;
        end
        default: begin
            state_d = Idle;
        end
    endcase
end

endmodule
