module register_lock(
    input wire clk_i,
    input wire rst_ni,
    input wire rst_9,
    input wire jtag_unlock,
    output reg [5:0] reglk_mem
);

always @(posedge clk_i) begin
    if (~(rst_ni && ~jtag_unlock && ~rst_9)) begin
        for (int i = 0; i < 6; i++) begin
            reglk_mem[i] <= 1'b0;
        end
    end
    // Additional clock-edge driven logic should be added here if needed.
end

endmodule
