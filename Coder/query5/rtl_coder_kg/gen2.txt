module register_lock(
    input wire clk_i,
    input wire rst_ni,
    input wire rst_9,
    input wire jtag_unlock,
    output reg [5:0] reglk_mem
);

    always @(posedge clk_i) begin
        if (~(rst_ni && ~rst_9)) begin
            for (int j = 0; j < 6; j = j + 1) begin
                reglk_mem[j] <= 1'b0; 
            end
        end
    end

endmodule
