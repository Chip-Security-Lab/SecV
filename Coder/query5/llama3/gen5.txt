module register_lock(
    input wire clk_i,
    input wire rst_ni,
    input wire rst_9,
    input wire jtag_unlock,
    output reg [5:0] reglk_mem
);

// Enhanced Reset Logic
wire reset_condition = ~(rst_ni && ~jtag_unlock && ~rst_9);

always @(posedge clk_i) begin
    if (reset_condition) begin
        for (int i = 0; i < 6; i++) begin
            reglk_mem[i] <= 1'b0;
        end
    end
end

endmodule
