module glitchEx(
    input wire in0,
    input wire in1,
    input wire sel,
    output wire z
);
    wire not_sel;
    wire and_out1, and_out2;
    
    assign not_sel = ~sel;
    assign and_out1 = not_sel & in0;
    assign and_out2 = sel & in1;
    
    // Case 1: Standard Multiplexer Output Logic
    // assign z = and_out1 | and_out2;

    // Case 2: Extended Logic Output
    assign z = and_out1 | and_out2 | (in0 & in1);
endmodule
