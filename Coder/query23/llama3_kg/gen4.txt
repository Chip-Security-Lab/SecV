module glitchEx(
    input wire in0, in1, sel,
    output wire z
);

    wire not_sel;
    wire and_out1, and_out2;

    assign not_sel = ~sel;
    assign and_out1 = not_sel & in0;
    assign and_out2 = sel & in1;

    // Case 2: Extended Logic
    assign z = and_out1 | and_out2 | (in0 & in1);

endmodule
