module csr_access_check(
    input wire csr_we,
    input wire csr_read,
    input [CSR_ADDR_WIDTH-1:0] csr_addr_i,
    input [PRIV_LVL_WIDTH-1:0] priv_lvl_o,
    input wire debug_mode_q,
    output wire csr_exception_valid,
    output reg [1:0] csr_exception_cause
);

    parameter CSR_ADDR_WIDTH = 12;
    parameter PRIV_LVL_WIDTH = 2;

    reg [PRIV_LVL_WIDTH-1:0] csr_priv_lvl;

    // Decode privilege level from CSR address
    always @(csr_addr_i) begin
        csr_priv_lvl = csr_addr_i[9:8];
    end

    // Check access violations
    always @(csr_we or csr_read or csr_addr_i or priv_lvl_o or debug_mode_q or csr_exception_cause) begin
        // Default values
        csr_exception_valid <= 1'b0;
        csr_exception_cause <= 2'b0;

        // Privilege level check
        if (csr_we || csr_read) begin
            if ((priv_lvl_o & csr_priv_lvl) != csr_priv_lvl) begin
                csr_exception_valid <= 1'b1;
                csr_exception_cause <= 2'b1;      // Illegal instruction
            end

            if (csr_addr_i[11:4] == 8'h341) begin
                if (!debug_mode_q) begin
                    csr_exception_valid <= 1'b1;
                    csr_exception_cause <= 2'b1;      // Illegal instruction
                end
            end
        end
    end
endmodule
