module csr_access_check
#(
  parameter CSR_ADDR_WIDTH = 12,
  parameter PRIV_LVL_WIDTH = 2
)
(
  input wire csr_we,
  input wire csr_read,
  input wire [CSR_ADDR_WIDTH-1:0] csr_addr_i,
  input wire [PRIV_LVL_WIDTH-1:0] priv_lvl_i,
  input wire debug_mode_q,
  
  output wire csr_exception_valid,
  output wire [CSR_ADDR_WIDTH-1:0] csr_exception_cause
);
  
  assign csr_exception_valid = 0;
  assign csr_exception_cause = {CSR_ADDR_WIDTH{1'b0}};

  // Privilege level checking
  always@(*) begin
    if((csr_addr_i[9:8] > priv_lvl_i) || 
       (CSR_ADDR_WIDTH == 12 && csr_addr_i == 12'h341 && priv_lvl_i != 2'b11) || // Case 1: Restrictive Access
       //(CSR_ADDR_WIDTH == 12 && csr_addr_i == 12'h341) || // Case 2: Relaxed Access
       (csr_addr_i[11:10] == 2'b11 && priv_lvl_i != 2'b11)) 
    {
        csr_exception_valid = 1;
        csr_exception_cause = {CSR_ADDR_WIDTH{1'b1}};
    }
  end

  // Debug mode access
  always@(debug_mode_q) begin
    if(csr_addr_i[11:10] == 2'b10 && debug_mode_q == 1'b0) begin
        csr_exception_valid = 1;
        csr_exception_cause = {CSR_ADDR_WIDTH{1'b1}};
    end
  end
endmodule
