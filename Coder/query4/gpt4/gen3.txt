module csr_access_check
#(
  parameter CSR_ADDR_WIDTH = 12,
  parameter PRIV_LVL_WIDTH = 2
)
(
  input wire csr_we,
  input wire csr_read,
  input wire [CSR_ADDR_WIDTH-1:0] csr_addr_i,
  input wire [PRIV_LVL_WIDTH-1:0] priv_lvl_i,
  input wire debug_mode_q,
  output reg csr_exception_valid,
  output reg [31:0] csr_exception_cause
);

  always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
      csr_exception_valid <= 1'b0;
      csr_exception_cause <= 32'h0;
    end else if (csr_we | csr_read) begin
      // Default: No exception
      csr_exception_valid <= 1'b0;
      csr_exception_cause <= 32'h0;

      // Privilege level checking
      if ((csr_addr_i[9:8] > priv_lvl_i) && (csr_addr_i != 12'h341)) begin
        csr_exception_valid <= 1'b1;
        csr_exception_cause <= 32'hSomething; // Replace with actual exception code
      end

      // Debug mode access
      if (((csr_addr_i >= 12'h7C0) && (csr_addr_i < 12'h800)) && !debug_mode_q) begin
        csr_exception_valid <= 1'b1;
        csr_exception_cause <= 32'hSomething; // Replace with actual exception code
      end
    end
  end

endmodule
