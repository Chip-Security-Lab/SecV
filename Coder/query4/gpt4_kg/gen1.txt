module csr_access_check
#(
    parameter CSR_ADDR_WIDTH = 12,
    parameter PRIV_LVL_WIDTH = 2
)
(
    input wire csr_we,
    input wire csr_read,
    input wire [CSR_ADDR_WIDTH-1:0] csr_addr_i,
    input wire [PRIV_LVL_WIDTH-1:0] priv_lvl_o,
    input wire debug_mode_q,
    output wire csr_exception_valid,
    output wire [PRIV_LVL_WIDTH-1:0] csr_exception_cause
);
    // Exception cause encoding
    localparam ILLEGAL_INSTR = 2'b01;

    // CSR privilege level decoding
    wire [PRIV_LVL_WIDTH-1:0] csr_priv_lvl = csr_addr_i[9:8];

    always @* begin
        // Default values: no exception
        csr_exception_valid = 1'b0;        // No exception by default
        csr_exception_cause = 2'b00;      // Default cause is 0 (no exception)

        // Check if there is a read or write request to a CSR
        if (csr_we || csr_read) begin
            // Privilege level check: If the current privilege level does not match the CSR's required level
            if ((priv_lvl_o & csr_priv_lvl) != csr_priv_lvl) begin
                csr_exception_valid = 1'b1;        // Set exception as valid
                csr_exception_cause = ILLEGAL_INSTR; // Cause: illegal instruction
            end

            // Debug mode access check: If accessing debug-specific CSRs outside of debug mode
            if (csr_addr_i[11:4] == 8'h7b && !debug_mode_q) begin
                csr_exception_valid = 1'b1;        // Set exception as valid
                csr_exception_cause = ILLEGAL_INSTR; // Cause: illegal instruction
            end
        end
    end
endmodule
