module csr_access_check #(
  parameter CSR_ADDR_WIDTH = 12,
  parameter PRIV_LVL_WIDTH = 2
) (
  input wire csr_we,
  input wire csr_read,
  input wire [CSR_ADDR_WIDTH-1:0] csr_addr_i,
  input wire [PRIV_LVL_WIDTH-1:0] priv_lvl_o,
  input wire debug_mode_q,
  output wire csr_exception_valid,
  output wire [4:0] csr_exception_cause
);

  // Constants
  localparam PRIV_LVL_M = 2'b11; // Machine mode privilege level
  localparam PRIV_LVL_S = 2'b10; // Supervisor mode privilege level
  localparam PRIV_LVL_U = 2'b01; // User mode privilege level
  localparam PRIV_LVL_MASK = 2'b11; // Mask for privilege level bits

  // Debug-specific CSRs
  localparam CSR_ADDR_DBG_START = 12'h7B0;
  localparam CSR_ADDR_DBG_END = 12'h7BF;

  // MEPC CSR address
  localparam CSR_ADDR_MEPC = 12'h341;

  // Default values
  assign csr_exception_valid = 1'b0;
  assign csr_exception_cause = 5'b0;

  // Privilege level check
  wire priv_lvl_mismatch;
  assign priv_lvl_mismatch = (csr_we || csr_read) && (priv_lvl_o < csr_addr_i[9:8]);

  // Debug mode check
  wire debug_mode_violation;
  assign debug_mode_violation = (csr_we || csr_read) && (csr_addr_i >= CSR_ADDR_DBG_START) && (csr_addr_i <= CSR_ADDR_DBG_END) && !debug_mode_q;

  // MEPC access check
  wire mepc_access_violation;
  assign mepc_access_violation = (csr_we || csr_read) && (csr_addr_i == CSR_ADDR_MEPC) && (priv_lvl_o != PRIV_LVL_M);

  // Exception handling
  always @(*) begin
    if (priv_lvl_mismatch) begin
      csr_exception_valid = 1'b1;
      csr_exception_cause = 5'b00001; // Illegal instruction
    end else if (debug_mode_violation) begin
      csr_exception_valid = 1'b1;
      csr_exception_cause = 5'b00010; // Illegal instruction
    end else if (mepc_access_violation) begin
      csr_exception_valid = 1'b1;
      csr_exception_cause = 5'b00011; // Illegal instruction
    end else begin
      csr_exception_valid = 1'b0;
      csr_exception_cause = 5'b0;
    end
  end

endmodule
